cmake_minimum_required(VERSION 3.16)
project(primebot_backend VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HTTPLIB REQUIRED IMPORTED_TARGET cpp-httplib)
add_executable(primebot_backend src/main.cpp)
target_link_libraries(primebot_backend PRIVATE Threads::Threads nlohmann_json::nlohmann_json PkgConfig::HTTPLIB)
target_compile_options(primebot_backend PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME api_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/api_test.py $<TARGET_FILE:primebot_backend>)
