cmake_minimum_required(VERSION 3.16)
project(primebot_backend VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)

# nlohmann/json：优先用系统包（Ubuntu apt），Windows 走 third_party 单头文件
find_package(nlohmann_json 3.2 QUIET)
if(NOT nlohmann_json_FOUND)
  set(JSON_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/json/single_include")
  if(NOT EXISTS "${JSON_INCLUDE}/nlohmann/json.hpp")
    message(FATAL_ERROR "缺少 nlohmann/json，先运行 bash scripts/fetch_deps.sh")
  endif()
  add_library(nlohmann_json INTERFACE)
  add_library(nlohmann_json::nlohmann_json ALIAS nlohmann_json)
  target_include_directories(nlohmann_json INTERFACE "${JSON_INCLUDE}")
endif()

# cpp-httplib：Linux 有 pkg-config 就走系统包，否则用 third_party 的头文件版
find_package(PkgConfig QUIET)
if(PKG_CONFIG_FOUND)
  pkg_check_modules(HTTPLIB QUIET IMPORTED_TARGET cpp-httplib)
endif()
if(TARGET PkgConfig::HTTPLIB)
  set(HTTPLIB_TARGET PkgConfig::HTTPLIB)
else()
  set(HTTPLIB_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/cpp-httplib")
  if(NOT EXISTS "${HTTPLIB_INCLUDE}/httplib.h")
    message(FATAL_ERROR "缺少 cpp-httplib，先运行 bash scripts/fetch_deps.sh")
  endif()
  add_library(httplib INTERFACE)
  target_include_directories(httplib INTERFACE "${HTTPLIB_INCLUDE}")
  set(HTTPLIB_TARGET httplib)
endif()

add_executable(primebot_backend src/main.cpp)
target_link_libraries(primebot_backend PRIVATE Threads::Threads nlohmann_json::nlohmann_json ${HTTPLIB_TARGET})
if(MSVC)
  target_compile_options(primebot_backend PRIVATE /W3 /utf-8)
else()
  target_compile_options(primebot_backend PRIVATE -Wall -Wextra -Wpedantic)
endif()

enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME api_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/api_test.py $<TARGET_FILE:primebot_backend>)
